Google Apps Script (`Code.gs`) สำหรับแปลง Google Sheets เป็น JSON API

หากคุณต้องการดึงข้อมูลด้วย Apps Script API แทน CSV Published URL ให้คัดลอกโค้ดด้านล่างนี้ไปวางในส่วนขยาย **Extensions -> Apps Script** บน Google Sheets ของคุณ: 
function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var result = [];

  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }

  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
